  sourceConnector string
  textProcessorId string fk
}

  id string pk
  normalizationRules "List<Rule>"
  tokenizer string
}

  id string pk
  content string
  metadata Map
  ingestionServiceId string fk
  repositoryId string fk
}

document_repositories [color: blue] {
  id string pk
  database string
}

indexer_services [color: purple] {
  id string pk
  indexVersion string
  textProcessorId string fk
}

inverted_indexes [color: purple] {
  id string pk
  indexerId string fk
  termStatistics Map
}

index_shards [color: purple] {
  id string pk
  invertedIndexId string fk
  documentRange Range
  searchShardId string fk
}

users [color: green] {
  id string pk
  username string
  email string
}

queries [color: green] {
  id string pk
  userId string fk
  text string
  topK int
  coordinatorId string fk
}

search_coordinators [color: green] {
  id string pk
}

search_shards [color: green] {
  id string pk
  coordinatorId string fk
  endpoint string
}

query_caches [color: orange] {
  id string pk

  cacheStore string
  ttl Duration
}

search_results [color: orange] {
  id string pk
  queryId string fk
  documentId string fk
  score double
  snippet string
}

  rankingWeights Map
